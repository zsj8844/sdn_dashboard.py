## 1. 基础网络功能

### 1.1 OpenFlow 1.3 协议支持

#### 功能概述
实现完整的 OpenFlow 1.3 协议支持，包括流表下发、统计收集、多级流表管理等核心功能。

#### 技术实现

##### 1.1.1 流表下发
**实现位置**: `ble_switch_13.py:add_flow()`

**核心代码**:
```python
def add_flow(self, datapath, priority, match, actions, 
             idle_timeout=0, hard_timeout=0, table_id=0):
    """添加流表规则"""
    ofproto = datapath.ofproto
    parser = datapath.ofproto_parser
    
    inst = [parser.OFPInstructionActions(ofproto.OFPIT_APPLY_ACTIONS, actions)]
    mod = parser.OFPFlowMod(
        datapath=datapath,
        table_id=table_id,
        priority=priority,
        match=match,
        instructions=inst,
        idle_timeout=idle_timeout,
        hard_timeout=hard_timeout
    )
    datapath.send_msg(mod)
    self.logger.info(f"流表添加成功 - 表ID:{table_id}, 优先级:{priority}")
```

**参数说明**:
- `datapath`: 交换机数据路径对象
- `priority`: 流表优先级（0-65535）
- `match`: 流表匹配条件（OFPMatch对象）
- `actions`: 流表动作列表
- `idle_timeout`: 空闲超时时间（秒）
- `hard_timeout`: 硬超时时间（秒）
- `table_id`: 流表ID（0-254）

```python
def delete_flow(self, datapath, match=None, priority=None, table_id=0):
    """删除流表规则"""
    ofproto = datapath.ofproto
    parser = datapath.ofproto_parser
    
    if match is None:
        match = parser.OFPMatch()
    
    mod = parser.OFPFlowMod(
        datapath=datapath,
        table_id=table_id,
        priority=priority if priority is not None else ofproto.OFP_DEFAULT_PRIORITY,
        match=match,
        command=ofproto.OFPFC_DELETE,
        out_port=ofproto.OFPP_ANY,
        out_group=ofproto.OFPG_ANY
    )
    datapath.send_msg(mod)
    self.logger.info(f"流表删除成功 - 表ID:{table_id}")
```

### 1.2 拓扑管理

#### 功能概述
基于 LLDP 协议自动发现网络设备和链路，并维护拓扑数据结构。

#### 技术实现

##### 1.2.1 LLDP 数据包发送
**实现位置**: `ble_switch_13.py:_lldp_sender()`

**核心代码**:
```python
def _lldp_sender(self):
    """LLDP数据包发送线程"""
    while True:
        hub.sleep(self.lldp_interval)
        for dpid, datapath in list(self.topology['switches'].items()):
            try:
                self.send_lldp_packets(datapath)
            except Exception as e:
                self.logger.warning(f"发送LLDP到交换机 {dpid} 失败: {e}")
```

**LLDP 数据包结构**:
- **Chassis ID**: 交换机 MAC 地址
- **Port ID**: 端口号
- **TTL**: 10 秒
- **System Name**: 交换机名称 (switch-{dpid})

##### 1.2.2 LLDP 数据包处理
**实现位置**: `ble_switch_13.py:_handle_lldp_packet()`

**核心代码**:
```python
def _handle_lldp_packet(self, dpid, in_port, pkt):
    """处理收到的LLDP数据包"""
    lldp_pkt = pkt.get_protocol(lldp.lldp)
    if not lldp_pkt:
        return
    
    chassis_id = None
    port_id = None
    system_name = None
    
    for tlv in lldp_pkt.tlvs:
        if tlv.tlv_type == lldp.lldp.LLDP_TLV_CHASSIS_ID:
            chassis_id = tlv.chassis_id
        elif tlv.tlv_type == lldp.lldp.LLDP_TLV_PORT_ID:
            port_id = tlv.port_id
        elif tlv.tlv_type == lldp.lldp.LLDP_TLV_SYSTEM_NAME:
            system_name = tlv.system_name
    
    if chassis_id and port_id and system_name:
        src_switch = system_name.decode('utf-8')
        src_port = int(port_id.decode('utf-8'))
        dst_switch = f'switch-{dpid}'
        dst_port = in_port
        
        self._add_link(src_switch, src_port, dst_switch, dst_port)
```

**拓扑数据结构**:
```python
self.topology = {
    'switches': {},      # {dpid: datapath}
    'links': [],         # [{src_switch, src_port, dst_switch, dst_port, timestamp}]
    'hosts': {}          # {mac: {mac, ip, switch_dpid, port, first_seen}}
}
```

```python
class IoTExtensionManager:
    """IoT扩展管理器"""
    
    def __init__(self):
        self.field_classes = {
            1: SensorTypeField,
            2: DevicePriorityField,
            3: RouteSelectionField,
            4: AppDeploymentField,
            5: RoleSwitchField
        }
    
    def create_iot_extension(self, **kwargs):
        """创建IoT扩展消息"""
        fields = []
        
        if 'sensor_type' in kwargs:
            fields.append(SensorTypeField(kwargs['sensor_type']))
        if 'device_priority' in kwargs:
            fields.append(DevicePriorityField(kwargs['device_priority']))
        if 'route_selection' in kwargs:
            fields.append(RouteSelectionField(kwargs['route_selection']))
        if 'app_deployment' in kwargs:
            fields.append(AppDeploymentField(kwargs['app_deployment']))
        if 'role_switch' in kwargs:
            fields.append(RoleSwitchField(kwargs['role_switch']))
        
        return IoTExtensionMessage(fields=fields)
```

```python
def process_iot_extension(self, msg):
    """处理IoT扩展消息"""
    if not self.extension_enabled:
        return None
    
    try:
        extension = self.extension_manager.parse_message(msg)
        extension_data = self.extension_manager.get_field_dict(extension)
        
        self.logger.debug(f"处理IoT扩展: {extension_data}")
        return extension_data
    except Exception as e:
        self.logger.warning(f"处理IoT扩展失败: {e}")
        return None
```

## 4. 路径优化功能

### 4.1 智能路径优化

#### 功能概述
基于流量特征和扩展字段元数据识别应用类型，根据应用类型自动选择最优传输路径。

#### 技术实现
**实现位置**: `ble_switch_13.py:_packet_in_handler()`

**核心代码**:
```python
@set_ev_cls(ofp_event.EventOFPPacketIn, MAIN_DISPATCHER)
def _packet_in_handler(self, ev):
    """处理Packet-In消息"""
    msg = ev.msg
    datapath = msg.datapath
    ofproto = datapath.ofproto
    parser = datapath.ofproto_parser
    
    in_port = msg.match['in_port']
    pkt = packet.Packet(msg.data)
    eth = pkt.get_protocol(ethernet.ethernet)
    
    if eth.ethertype == ethernet.ETH_TYPE_LLDP:
        self._handle_lldp_packet(datapath.id, in_port, pkt)
        return
    
    if self.extension_enabled:
        extension_data = self.process_iot_extension(msg)
        if extension_data:
            self._apply_optimized_routing(datapath, in_port, pkt, extension_data)
            return
    
    super()._packet_in_handler(ev)

def _apply_optimized_routing(self, datapath, in_port, pkt, extension_data):
    """应用优化路由"""
    dpid = datapath.id
    
    if 'device_priority' in extension_data:
        priority = extension_data['device_priority']
        if priority >= 8:
            self.logger.info(f"应用高优先级路由 - 设备优先级: {priority}")
        elif priority >= 5:
            self.logger.info(f"应用中优先级路由 - 设备优先级: {priority}")
        else:
            self.logger.info(f"应用低优先级路由 - 设备优先级: {priority}")
    
    if 'route_selection' in extension_data:
        route_type = extension_data['route_selection']
        if route_type == 1:
            self.logger.info("选择低延迟路径")
        elif route_type == 2:
            self.logger.info("选择高带宽路径")
        else:
            self.logger.info("选择默认路径")
```

```python
def deploy_application(self, app_id, device_id):
    """部署应用到设备"""
    if app_id not in self.applications:
        raise ValueError(f"应用 {app_id} 不存在")
    
    app = self.applications[app_id]
    
    deployment_record = {
        'app_id': app_id,
        'device_id': device_id,
        'timestamp': datetime.now().isoformat(),
        'action': 'deploy',
        'status': 'success'

    return deployment_record
```

```python
def undeploy_application(self, app_id, device_id):
    """从设备卸载应用"""
    if app_id not in self.applications:
        raise ValueError(f"应用 {app_id} 不存在")
    
    app = self.applications[app_id]
    
    deployment_record = {
        'app_id': app_id,
        'device_id': device_id,
        'timestamp': datetime.now().isoformat(),
        'action': 'undeploy',
        'status': 'success'
    }
    
    if device_id in app.deployed_devices:
        app.deployed_devices.remove(device_id)
    
    if not app.deployed_devices:
        app.status = AppStatus.STOPPED
    
    self.deployment_history.append(deployment_record)
    return deployment_record
```

```python
def switch_mode(self, device_id, target_mode, force=False):
    """切换设备模式"""
    if device_id not in self.devices:
        raise ValueError(f"设备 {device_id} 不存在")
    
    device = self.devices[device_id]

    device.last_switch_time = datetime.now().isoformat()
    device.switch_count += 1
    
    switch_record = {
        'device_id': device_id,
        'old_mode': old_mode.value,
        'new_mode': target_mode.value,
        'timestamp': device.last_switch_time,
        'forced': force,
        'status': 'success'
    }
    
    self.switch_history.append(switch_record)
    return switch_record
```

**模式识别**:
```python
def identify_device_mode(self, device_id, traffic_pattern=None, 
                        resource_usage=None):
    """识别设备模式"""
    if device_id not in self.devices:
        raise ValueError(f"设备 {device_id} 不存在")
    
    device = self.devices[device_id]
    capabilities = device.capabilities
    
    if resource_usage:
        cpu_usage = resource_usage.get('cpu_usage', 0)
        memory_usage = resource_usage.get('memory_usage', 0)
        
        if cpu_usage > 70 or memory_usage > 70:
            if capabilities.supports_pure_forwarding:
                return DeviceMode.PURE_FORWARDING
    
    if traffic_pattern:
        traffic_volume = traffic_pattern.get('volume', 'low')
        traffic_type = traffic_pattern.get('type', 'data')
        
        if traffic_type == 'iot' and capabilities.supports_iot_gateway:
            return DeviceMode.IOT_GATEWAY
        
        if traffic_volume == 'high' and capabilities.supports_compute_hybrid:
            return DeviceMode.COMPUTE_HYBRID
    
    return device.current_mode
```
